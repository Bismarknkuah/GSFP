1) ALLOW PREVIOUS-DATE SUBMISSION
Find any validation like:

REMOVE it or replace with a 30-day window:
    const diffDays = (Date.now() - new Date(date))/(86400000);

2) EDIT & RESUBMIT (add this handler + route)
    // PATCH /api/reports/:id  — caterer edits own pending/rejected report
    exports.update = async (req, res) => {
      const Report = require('../models/Report');
      const r = await Report.findById(req.params.id);
      if (!r) return res.status(404).json({ error:'Report not found' });
      if (String(r.caterer_id) !== String(req.user._id||req.user.id))
        return res.status(403).json({ error:'Not your report' });
      if (!['pending','rejected'].includes(r.status))
        return res.status(400).json({ error:'Only pending or rejected reports can be edited' });
      const allowed = ['food_type','students_fed','time_ready','time_served','notes','date'];
      allowed.forEach(k => { if (req.body[k] !== undefined) r[k] = req.body[k]; });
      r.status = 'pending';                    // resubmission resets to pending
      r.headmaster_comment = null;
      r.resubmitted_at = new Date();
      await r.save();
      res.json({ report:r, resubmitted:true });
    };
Route:  router.patch('/:id', authenticate, catererOnly, c.update);
